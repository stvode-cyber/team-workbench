# v3.1.0 融合指南针 · C# WPF 大壳 × Python 多模型调度引擎

> **定位**：v3.0 C# WPF 只做了「AI 员工注册中心 + 状态看板」，不能让 11 个专岗**同时跑**。
> v3.1.0 补的是「执行引擎」——把 11 个专岗的 system_prompt 喂给调度引擎，
> 支持 DIRECT / SEQUENTIAL / PARALLEL / HYBRID 四种调度模式 + 多模型路由 + durable execution。
> 
> **本文档是 v3.1.0 的唯一设计文档**。所有开发、测试、发版都以这里为准。
> 本文档不替代 AGENTS.md——AGENTS.md 管项目设计层（业务方向/技术栈/目录约定），
> 本文档管 v3.1.0 这个版本的完整实现蓝图。

## ① 融合本质（核心洞察，必须先读）

```
现有 v3.0 C# WPF ──► 管「谁」（11 个专岗的 system_prompt + 职责 + 边界）
                         │
                         │  v3.1.0 融合点：专岗 system_prompt 自动流入调度引擎
                         │
                    Python Scheduler ──► 管「怎么让它们同时跑」（调度模式 + 多模型 + durable execution）
```

**两个项目是垂直关系，不是并列关系：**
- C# WPF 是「注册中心」，负责定义 11 个专岗 + 托盘/热键/浮动球/原生体验
- Python Scheduler 是「执行引擎」，负责调度这些专岗 + 跑 LLM + checkpoint

**为什么不整合成一个技术栈？**
- C# WPF 做桌面壳（托盘/热键/浮动球）是它的地盘
- Python FastAPI + LiteLLM 是多模型调度的事实标准（CustomLLM、LiteLLM router、LangGraph 都在 Python 生态）
- 独立进程 + localhost HTTP/SSE = 两个进程各挂各的，互不影响

## ④ 数据库 Schema（一个 SQLite，WAL 模式）

```sql
-- === PRAGMA（启动时必须设置）===
PRAGMA journal_mode=WAL;           -- 多进程并发读写安全
PRAGMA synchronous=NORMAL;         -- WAL 模式下性能和安全的最佳平衡
PRAGMA busy_timeout=5000;          -- 另一进程锁了 DB，等 5s

-- === 现有表（v3.0 C# 已有，不动）===
-- agents, install_log, status_events

-- === 🆕 Python Scheduler 表 ===

-- sessions：每次调度会话
CREATE TABLE IF NOT EXISTS sessions (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    session_id TEXT NOT NULL UNIQUE,        -- UUID v4
    name TEXT,                               -- 用户给会话起的名
    mode TEXT NOT NULL,                      -- DIRECT / SEQUENTIAL / PARALLEL / HYBRID
    status TEXT NOT NULL DEFAULT 'pending',  -- pending / running / paused / completed / failed
    input_json TEXT,                         -- 会话输入（prompt + 上下文）
    output_json TEXT,                        -- 最终输出
    error_message TEXT,
    started_at TEXT,
    completed_at TEXT,
    created_at TEXT NOT NULL
);

-- agent_runs：每个 session 里每个 agent 的执行记录
CREATE TABLE IF NOT EXISTS agent_runs (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    session_id TEXT NOT NULL REFERENCES sessions(session_id),
    agent_name TEXT NOT NULL,                -- drize-engineer
    step_index INTEGER NOT NULL,             -- 在 session 里的序号
    wave INTEGER NOT NULL DEFAULT 0,         -- DAG wave：同 wave 并行
    input_json TEXT,                         -- 传给这个 agent 的完整 input
    output_json TEXT,                        -- 这个 agent 的原始输出
    llm_provider TEXT,                       -- openrouter / ollama / fallback
    llm_model TEXT,                          -- gpt-4o-mini / qwen2.5:7b
    status TEXT NOT NULL,                    -- pending / running / completed / failed / skipped / approval_pending
    error_message TEXT,
    started_at TEXT,
    completed_at TEXT,
    duration_ms INTEGER,
    retry_count INTEGER DEFAULT 0,
    output_hash TEXT,                        -- SHA256(output_json)，用于幂等
    idempotency_key TEXT NOT NULL,           -- SHA256(session_id + step_index + input_hash)
    UNIQUE(session_id, step_index)
);

-- checkpoints：durability（MeshFlow 生产模式）
-- 每个 step：INSERT STARTED → 执行 → INSERT COMMITTED
-- 恢复时：STARTED 无 COMMITTED → 重跑；COMMITTED → 跳过
CREATE TABLE IF NOT EXISTS checkpoints (
    session_id TEXT NOT NULL REFERENCES sessions(session_id),
    step_id TEXT NOT NULL,                   -- = agent_runs.id 的字符串
    step_index INTEGER NOT NULL,
    status TEXT NOT NULL,                    -- STARTED / COMMITTED / SKIPPED
    input_hash TEXT,
    output_hash TEXT,
    output_data BLOB,                        -- gzip 压缩的 JSON
    started_at TEXT,
    committed_at TEXT,
    idempotency_key TEXT NOT NULL,
    PRIMARY KEY (session_id, step_id)
);

-- pipelines：用户保存的 DAG 模板
CREATE TABLE IF NOT EXISTS pipelines (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    pipeline_id TEXT NOT NULL UNIQUE,
    name TEXT NOT NULL,
    description TEXT,
    dag_json TEXT NOT NULL,                  -- {"nodes":[...], "edges":[...]}
    created_at TEXT NOT NULL,
    updated_at TEXT NOT NULL
);

-- approval_requests：HITL 中断审批门
CREATE TABLE IF NOT EXISTS approval_requests (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    session_id TEXT NOT NULL REFERENCES sessions(session_id),
    run_id INTEGER NOT NULL REFERENCES agent_runs(id),
    reason TEXT NOT NULL,                    -- 为什么要审批
    request_payload TEXT,                    -- 等审批的操作（JSON）
    status TEXT NOT NULL DEFAULT 'pending',  -- pending / approved / rejected / timeout
    approver TEXT,
    response_at TEXT,
    expires_at TEXT NOT NULL,                -- 审批超时时间（默认 30 分钟）
    created_at TEXT NOT NULL
);

-- llm_config：每个 agent 绑定的 LLM 配置（可覆盖全局默认）
CREATE TABLE IF NOT EXISTS llm_config (
    agent_name TEXT PRIMARY KEY,             -- drize-engineer
    provider TEXT NOT NULL,                  -- openrouter / ollama / mock
    model TEXT NOT NULL,                     -- gpt-4o-mini / qwen2.5:7b
    temperature REAL DEFAULT 0.7,
    max_tokens INTEGER DEFAULT 4096,
    fallback_provider TEXT,                  -- 失败降级
    fallback_model TEXT,
    updated_at TEXT NOT NULL
);

-- audit_log：全量审计（谁/什么时候/改了什么）
CREATE TABLE IF NOT EXISTS audit_log (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    action TEXT NOT NULL,                    -- SESSION_CREATE / RUN_COMPLETE / APPROVAL_APPROVED
    target_type TEXT NOT NULL,               -- session / run / checkpoint
    target_id TEXT NOT NULL,
    metadata_json TEXT,
    actor TEXT,                              -- user / system / csharp_host
    created_at TEXT NOT NULL
);

-- 索引（高频查询）
CREATE INDEX IF NOT EXISTS idx_agent_runs_session ON agent_runs(session_id);
CREATE INDEX IF NOT EXISTS idx_agent_runs_status ON agent_runs(status);
CREATE INDEX IF NOT EXISTS idx_checkpoints_session ON checkpoints(session_id);
CREATE INDEX IF NOT EXISTS idx_audit_log_time ON audit_log(created_at);
```

## ⑥ Scheduler 核心逻辑（4 种模式）

### DIRECT（单 Agent 直调）
```python
async def _run_direct(self, agents, prompt):
    run = await self._create_run(agents[0], step_index=0, wave=0)
    await self.checkpointer.save_step_started(run.id, run.idempotency_key)
    output = await self.llm.acompletion(
        model=run.llm_model,
        messages=[{"role": "system", "content": run.system_prompt},
                  {"role": "user", "content": prompt}],
        stream=True
    )
    await self.checkpointer.save_step_committed(run.id, output, sha256(output))
    yield SseEvent(event="agent_completed", data={...})
```

### SEQUENTIAL（Agent A 输出 = Agent B 输入）
```python
async def _run_sequential(self, agents, prompt):
    prev_output = prompt
    for i, agent in enumerate(agents):
        yield SseEvent(event="agent_started", data={step_index=i, wave=i})
        run = await self._create_run(agent, step_index=i, wave=i)
        # 关键：Agent B 的 input 包含 Agent A 的 output
        messages = [
            {"role": "system", "content": run.system_prompt},
            {"role": "user", "content": prev_output},  # ← 上一个的输出当下一个的输入
        ]
        output = await self.llm.acompletion(model=run.llm_model, messages=messages, stream=True)
        prev_output = output  # ← 传下去
        await self.checkpointer.save_step_committed(run.id, output, sha256(output))
        yield SseEvent(event="agent_completed", data={run_id: run.id})
```

### PARALLEL（asyncio.gather 同时跑）
```python
async def _run_parallel(self, agents, prompt):
    # 所有 agent 同时跑，各自 output 独立
    tasks = [self._run_single_agent(agent, step_index=i, wave=0, prompt=prompt)
             for i, agent in enumerate(agents)]
    results = await asyncio.gather(*tasks, return_exceptions=True)
    for i, result in enumerate(results):
        if isinstance(result, Exception):
            yield SseEvent(event="agent_failed", data={agent_name: agents[i].name, error: str(result)})
```

### HYBRID ★（DAG 拓扑排序 + wave-based dispatch）
```python
async def _run_hybrid(self, agents, layers, initial_prompt):
    """
    layers = [[0, 1], [2], [3, 4]]  # graphlib.TopologicalSorter 分层
    Layer 0: node 0,1 → 同波 asyncio.gather 并行
    Layer 1: node 2 → 等 Layer 0 全完再跑
    Layer 2: node 3,4 → 同波并行
    """
    context = {"initial_prompt": initial_prompt}
    
    for wave_idx, layer in enumerate(layers):
        tasks = []
        for node_idx in layer:
            agent = agents[node_idx]
            task = self._run_single_agent(
                agent,
                step_index=node_idx,
                wave=wave_idx,
                context=context
            )
            tasks.append(task)
        
        # 同波并行
        results = await asyncio.gather(*tasks, return_exceptions=True)
        
        # 波间：收集 output 注入 context → 下一波用
        for result in results:
            if not isinstance(result, Exception):
                context[result.agent_name] = result.output  # Agent A 输出 → Agent B 能读
```

### Retry 指数退避
```python
async def _with_retry(self, coro, max_retries=3):
    last_exception = None
    for attempt in range(max_retries):
        try:
            return await asyncio.wait_for(coro, timeout=60)
        except Exception as e:
            last_exception = e
            if attempt < max_retries - 1:
                wait = 2 ** attempt  # 1s, 2s, 4s
                await asyncio.sleep(wait)
    raise last_exception  # 3 次全失败 → agent_runs.status=failed
```

### SseClient.cs（.NET 10 HttpClient + SSE 协议）

```csharp
// Services/SseClient.cs

public class SseClient : IAsyncDisposable
{
    private readonly HttpClient _http = new() { Timeout = Timeout.InfiniteTimeSpan };
    private CancellationTokenSource? _cts;
    private string _lastEventId = "";
    private int _retryDelayMs = 3000;
    
    public event EventHandler<SseEvent>? OnEvent;
    public event EventHandler? OnHeartbeat;
    
    public async Task ConnectAsync(Uri endpoint, HttpContent content)
    {
        _cts = new CancellationTokenSource();
        while (!_cts.IsCancellationRequested)
        {
            try
            {
                using var request = new HttpRequestMessage(HttpMethod.Post, endpoint);
                request.Content = content;
                request.Headers.TryAddWithoutValidation("Accept", "text/event-stream");
                if (!string.IsNullOrEmpty(_lastEventId))
                    request.Headers.TryAddWithoutValidation("Last-Event-ID", _lastEventId);
                
                using var resp = await _http.SendAsync(
                    request, HttpCompletionOption.ResponseHeadersRead, _cts.Token
                );
                resp.EnsureSuccessStatusCode();
                
                await using var stream = await resp.Content.ReadAsStreamAsync(_cts.Token);
                using var reader = new StreamReader(stream);
                await ParseSseLoopAsync(reader);
            }
            catch (OperationCanceledException) { return; }
            catch (Exception ex)
            {
                // 断线重连：指数退避，最大 30s
                await Task.Delay(_retryDelayMs, _cts.Token);
                _retryDelayMs = Math.Min(_retryDelayMs * 2, 30_000);
            }
        }
    }
    
    /// <summary>
    /// SSE 协议解析（EventSource 类在 WPF 里不可用，手动实现）:
    ///   : ping          → 心跳注释（忽略，触发 OnHeartbeat）
    ///   event: xxx      → 事件类型
    ///   data: json      → 一行或多行 JSON 负载
    ///   id: xxx         → 事件 ID（断线重连用 Last-Event-ID）
    ///   retry: ms       → 服务端建议的重连间隔
    ///   空行            → 一个 event 结束，触发 OnEvent
    /// </summary>
    private async Task ParseSseLoopAsync(StreamReader reader)
    {
        var current = new SseEvent();
        string? line;
        while ((line = await reader.ReadLineAsync()) != null)
        {
            if (_cts?.IsCancellationRequested ?? true) return;
            
            if (line.StartsWith(':')) { OnHeartbeat?.Invoke(this, EventArgs.Empty); continue; }
            if (string.IsNullOrWhiteSpace(line))
            {
                if (current.Data != null)
                {
                    _lastEventId = current.Id ?? _lastEventId;
                    OnEvent?.Invoke(this, current);
                }
                current = new SseEvent();
                continue;
            }
            
            var colon = line.IndexOf(':');
            if (colon < 0) continue;
            var field = line[..colon].Trim();
            var value = line[(colon + 1)..].Trim();
            
            switch (field)
            {
                case "event": current.EventType = value; break;
                case "data":  current.Data = current.Data == null ? value : current.Data + "\n" + value; break;
                case "id":    current.Id = value; break;
                case "retry": int.TryParse(value, out _retryDelayMs); break;
            }
        }
    }
}
```

### PythonEngineHost.cs（Windows subprocess 管理）

```csharp
// Services/PythonEngineHost.cs

public class PythonEngineHost : IAsyncDisposable
{
    private Process? _python;
    private readonly string _exePath;
    private readonly string _workingDir;
    private readonly Uri _healthEndpoint = new("http://127.0.0.1:8765/health");
    private Timer? _healthTimer;
    
    public async Task StartAsync()
    {
        var psi = new ProcessStartInfo
        {
            FileName = _exePath,
            Arguments = "main.py --host 127.0.0.1 --port 8765",
            WorkingDirectory = _workingDir,
            UseShellExecute = false,
            CreateNoWindow = true,
            RedirectStandardError = true,
            // .NET 10 新增：给 Python 自己的进程组
            // Ctrl+C 不会被 C# 主进程吞
            // Python 内部通过 CTRL_BREAK_EVENT 处理
        };
        
        _python = Process.Start(psi) ?? throw new InvalidOperationException("Python start failed");
        
        // 等就绪：最多 30s
        for (int i = 0; i < 30; i++)
        {
            await Task.Delay(1000);
            if (await IsHealthyAsync()) return;
        }
        throw new InvalidOperationException("Python engine failed to start within 30s");
    }
    
    public async Task StopAsync()
    {
        if (_python == null) return;
        
        // Step 1: 优雅关闭（让 Python 保存 checkpoint）
        try
        {
            using var client = new HttpClient { Timeout = TimeSpan.FromSeconds(3) };
            await client.PostAsync("http://127.0.0.1:8765/shutdown", null);
            await Task.Delay(1000);
        } catch { /* 端口可能已关 */ }
        
        // Step 2: 信号终止
        // .NET 10: _python.Signal(SignalReason.ControlCEvent)
        _python.CancelKeyPress += (s, e) => { e.Cancel = true; };
        _python.Kill(false);  // false = 给进程时间处理
        
        // Step 3: 5s 后强杀
        if (!_python.WaitForExit(5000))
        {
            _python.Kill(true);  // true = 杀整个进程树
        }
    }
    
    private async Task<bool> IsHealthyAsync()
    {
        try
        {
            using var client = new HttpClient { Timeout = TimeSpan.FromSeconds(2) };
            var resp = await client.GetAsync(_healthEndpoint);
            return resp.IsSuccessStatusCode;
        }
        catch { return false; }
    }
    
    public async ValueTask DisposeAsync()
    {
        _healthTimer?.Dispose();
        if (_python != null) await StopAsync();
        await CastManagedToDisposable(_python)?.DisposeAsync() ?? ValueTask.CompletedTask;
    }
}
```
